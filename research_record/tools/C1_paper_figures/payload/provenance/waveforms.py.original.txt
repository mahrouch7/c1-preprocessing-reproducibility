"""Display-only revision from frozen saved waveforms; no filtering, fitting or metrics."""
from pathlib import Path
from datetime import datetime,timezone,timedelta
import argparse,hashlib,json,os,sys,subprocess

def main():
    p=argparse.ArgumentParser(description=__doc__)
    p.add_argument('--project-root',type=Path,required=True)
    p.add_argument('--numeric',type=Path,required=True)
    p.add_argument('--output',type=Path,required=True)
    p.add_argument('--graphics-dependencies',type=Path,required=True)
    p.add_argument('--poppler',type=Path,required=True)
    a=p.parse_args()
    gate=(a.project_root/'research_record/state/C1_quality_gate.md').read_text(encoding='utf-8')
    charter=(a.project_root/'research_record/state/C1_editorial_charter.md').read_text(encoding='utf-8')
    assert 'C1-D116' in gate and 'C1-D108' in charter
    sys.path.insert(0,str(a.graphics_dependencies.resolve()))
    os.environ['MPLCONFIGDIR']=str(a.output.parent/'mplconfig')
    import numpy as np
    import matplotlib
    matplotlib.use('Agg')
    import matplotlib.pyplot as plt
    from matplotlib.lines import Line2D
    from matplotlib.ticker import MaxNLocator
    sha=lambda path:hashlib.sha256(path.read_bytes()).hexdigest()
    summary=json.loads((a.numeric/'summary.json').read_text(encoding='utf-8'))
    source=a.numeric/'waveforms.npz'
    assert sha(source)==summary['waveforms_sha256']=='13449ff67c61793b55062d23a91ee7ee21f1fceecad5c8868972f2f9d6c7d31f'
    assert not summary['synthetic_fixture']
    a.output.mkdir(parents=True,exist_ok=False)
    plt.rcParams.update({'font.family':'DejaVu Sans','font.size':7.4,'axes.labelsize':7.5,
     'xtick.labelsize':7,'ytick.labelsize':7,'pdf.fonttype':42,'svg.fonttype':'none',
     'svg.hashsalt':'C1AUD08CompactV3','path.simplify':False,'axes.spines.top':False,
     'axes.spines.right':False,'axes.linewidth':.65,'grid.linewidth':.35})
    with np.load(source,allow_pickle=False) as saved:
        clean=saved['clean'];noisy=saved['noisy'];processed=saved['processed']
        time=saved['time_after_onset_s']*1000
        assert clean.shape==(8000,6) and processed.shape==(3,3,8000,6)
        assert list(saved['conditions'])==['lowfreq__10','impulsive__10','mixed_gain_additive__10']
        assert list(saved['treatments'])==['none','median','savgol']
        assert np.array_equal(noisy,processed[:,0])
        colors=['#777777','#0072B2','#D55E00']
        names=['Low-frequency','Impulsive','Mixed']
        display=[]
        fig,axes=plt.subplots(3,3,figsize=(7.15,4.4))
        fig.subplots_adjust(left=.09,right=.985,top=.765,bottom=.13,hspace=.64,wspace=.19)
        for row,(ci,channel) in enumerate([(0,0),(1,3),(2,0)]):
            if ci==0:
                start,stop,anchor=0,1000,None
                rule='Fixed first 25-ms interval [0,25) after fault onset, same for current and voltage; retained-window boundary included.'
            elif ci==1:
                nonzero=np.flatnonzero(noisy[ci,:,channel]!=clean[:,channel])
                assert len(nonzero)
                anchor=int(nonzero[0]);start=max(0,min(anchor-40,8000-80));stop=start+80
                rule='2-ms interval centered on first injected pulse in this channel; clipped to retained window.'
            else:
                anchor=int(np.argmax(np.abs(noisy[ci,:,channel]-clean[:,channel])))
                start=max(0,min(anchor-160,8000-320));stop=start+320
                rule='8-ms interval centered on largest absolute corrupted-minus-clean deviation in this channel; clipped to retained window. No filter output or classification result selects the interval.'
            interval=slice(start,stop)
            views=[noisy[ci],processed[ci,1],processed[ci,2]]
            ymin=min(float(v[interval,channel].min()/1000) for v in [clean]+views)
            ymax=max(float(v[interval,channel].max()/1000) for v in [clean]+views)
            pad=(ymax-ymin)*.075 or .1
            for col,view in enumerate(views):
                ax=axes[row,col]
                ax.plot(time[interval],view[interval,channel]/1000,color=colors[col],lw=1.1,zorder=2)
                ax.plot(time[interval],clean[interval,channel]/1000,color='#111111',lw=.9,
                        linestyle=(0,(3.2,2.1)),zorder=3)
                ax.set_xlim(time[start],time[stop-1]);ax.set_ylim(ymin-pad,ymax+pad)
                ax.xaxis.set_major_locator(MaxNLocator(3));ax.yaxis.set_major_locator(MaxNLocator(3))
                ax.grid(alpha=.21);ax.tick_params(length=2.5,pad=2)
                if col:ax.tick_params(labelleft=False)
                else:ax.set_ylabel(r'$I_A$ (kA)' if channel==0 else r'$V_A$ (kV)')
            axes[row,0].set_title(f'({chr(97+row)}) {names[ci]}, 10 dB',loc='left',fontsize=7.8,pad=4)
            display.append({'group':chr(97+row),'condition':str(saved['conditions'][ci]),'channel':channel,
                'start_sample':start,'stop_sample_exclusive':stop,'anchor_sample':anchor,
                'interval_start_ms':float(time[start]),'interval_stop_exclusive_ms':stop*.025,
                'zoom_rule':rule,'same_interval_all_three_treatments':True,'same_vertical_limits_within_row':True})
        for col,title in enumerate(['Unfiltered','Median filter','Savitzky-Golay']):
            box=axes[0,col].get_position();fig.text((box.x0+box.x1)/2,.855,title,
                ha='center',fontsize=9,color=colors[col],fontweight='bold')
        fig.suptitle('Waveform processing: separate treatments and time-domain zooms',fontsize=9.8,y=.99)
        fig.legend(handles=[Line2D([],[],color='#111111',ls=(0,(3.2,2.1)),lw=.9,label='Clean reference (repeated in every panel)'),
                            Line2D([],[],color='#555555',lw=1.1,label='Signal for the indicated treatment')],
                   loc='upper center',bbox_to_anchor=(.5,.936),ncol=2,frameon=False,fontsize=7.6)
        fig.text(.535,.035,'Time after fault onset (ms)',ha='center',fontsize=8.2)
        stem=a.output/'waveform_processing_examples_compact_v3'
        for ext in ['pdf','svg','png']:
            metadata={'CreationDate':None,'ModDate':None} if ext=='pdf' else ({'Date':None} if ext=='svg' else None)
            fig.savefig(stem.with_suffix('.'+ext),dpi=200,facecolor='white',metadata=metadata)
        plt.close(fig)
    rendered=a.output/'actual_pdf_render'
    subprocess.run([str(a.poppler),'-singlefile','-r','200','-png',str(stem.with_suffix('.pdf')),str(rendered)],check=True,capture_output=True)
    record={'status':'DISPLAY_REVISION_GENERATED_ACTUAL_PDF_VISUAL_REVIEW_PENDING',
     'created_local':datetime.now(timezone(timedelta(hours=1))).isoformat(timespec='seconds'),
     'source_waveforms_sha256':sha(source),'source_summary_sha256':sha(a.numeric/'summary.json'),
     'data_processing_recomputed':False,'model_training_or_metrics_computed':False,
     'native_samples_preserved':True,'same_acquisition':summary['selection'],
     'display_groups':display,'notes':['Each colored trace has its own panel with a black dashed clean reference.',
       'Channels alternate by mechanism: low-frequency current A, impulsive voltage A, mixed current A. Companion V2 paired views retained internally.',
       'Mixed zoom is selected on the largest corruption deviation, not representative performance.',
       'All comparisons within a row share identical time and amplitude scales.',
       'Coincidence with the clean reference remains genuine; no trace offset or synthetic separation.'],
     'outputs':{p.name:sha(p) for p in a.output.iterdir() if p.is_file()}}
    (a.output/'display_receipt.json').write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8')
    print(json.dumps({'status':record['status'],'groups':display,'source_waveforms_sha256':sha(source)}))

if __name__=='__main__':main()
