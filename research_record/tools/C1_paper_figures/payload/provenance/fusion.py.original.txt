def save(fig,name,sources):

def average_ranks(values):
    a=np.asarray(values);order=np.argsort(a,kind='stable');result=np.empty(len(a));i=0
    while i<len(a):
        j=i+1
        while j<len(a) and a[order[j]]==a[order[i]]:j+=1
        result[order[i:j]]=(i+j-1)/2+1;i=j
    return result

    allpreps=['none']+PREPS;allcolors=['#333333']+PCOLORS;markers=['o','s','^','D','P'];points=[];rho={}
    for ax,n,title in zip(axes[1:],['lowfreq','mixed_gain_additive'],['(b) Low-frequency','(c) Mixed']):
        xs=[];ys=[]
        for p,c in zip(allpreps,allcolors):
            for s,marker in zip(SNRS,markers):
                condition=f'{n}__{s}'
                x=median(float(r['median_tpi_noisy']) for r in data if r['operator']==operator and r['preprocessor']==p and r['condition']==condition)
                vals=[float(r['macro_f1'])*100 for r in scores if r['study']=='robustness' and r['preprocessor']==p and r['protocol']=='deployment' and r['condition']==condition and r['representation']=='MULTI18'];assert len(vals)==8
                y=median(vals);assert x>0
                ax.scatter(x,y,c=c,marker=marker,s=20,linewidths=.7,zorder=3);xs.append(x);ys.append(y)
                points.append({'noise':n,'snr':s,'preprocessor':p,'median_channel_tpi':x,'median_classifier_macro_f1_pct':y})
        rho[n]=float(np.corrcoef(average_ranks(xs),average_ranks(ys))[0,1])
        if n=='mixed_gain_additive':
            from matplotlib.ticker import NullFormatter
            ax.set_xscale('log');ax.xaxis.set_minor_formatter(NullFormatter());ax.set_xlabel('TPI (log scale)')
        else:
            ax.locator_params(axis='x',nbins=3);ax.set_xlabel('TPI')
        ax.set_ylim(0,100);ax.set_yticks([0,50,100]);ax.set_title(title);ax.grid(axis='y',color='#dddddd');ax.set_ylabel('Macro-F1 (%)')
    with (ROOT/'tpi_association_points_verified.csv').open('w',encoding='utf-8',newline='') as f:
        writer=csv.DictWriter(f,fieldnames=list(points[0]),lineterminator='\n');writer.writeheader();writer.writerows(points)
    expected={r['noise']:float(r['spearman_rho']) for r in rows_from_external('tpi_exploratory_association.csv') if r['operator']==operator}
    assert all(abs(rho[k]-expected[k])<1e-12 for k in rho),(rho,expected)
    handles=[Line2D([],[],color=c,label=n,lw=1.3) for c,n in zip(allcolors,['None']+PNAMES)]
    handles += [Line2D([],[],ls='',marker=m,color='black',label=str(s)+' dB',ms=4) for m,s in zip(markers,SNRS)]
